            m <- m;  // return from Message.GetMessage

package CPA {
    params {
        prf: fn Bits(256), Bits(256) -> Bits(256)
    }

    import oracles {
        GetMessage(m: Bits(256)) -> Bits(256),
        Enc(k: Bits(256), m: Bits(256)) -> (Bits(256), Bits(256)),
    }

    state {
        k: Maybe(Bits(256))
    }

    oracle Enc(m: Bits(256)) -> (Bits(256), Bits(256)) {
        if (k == None) {
            k_ <-$ Bits(256) sample-name k;
            k <- Some(k_);
        }
        m <- invoke GetMessage(m);
        c <- invoke Enc(Unwrap(k), m);
        return c;
    }
}

package Construction {
    params {
        prf: fn Bits(256), Bits(256) -> Bits(256),
        xor: fn Bits(256), Bits(256) -> Bits(256)
    }

    oracle Enc(k: Bits(256), m: Bits(256)) -> (Bits(256), Bits(256)) {
        nonce <-$ Bits(256) sample-name nonce;
        return (nonce, xor(prf(k, nonce), m));
    }
}
package Message {
    params {
        bmsg: Bool
    }

    oracle GetMessage(m: Bits(256)) -> Bits(256) {
        if bmsg {
            return 0_256;
        } else {
            return m;
        }
    }
}
package Nonce {
    params {
        bnonce: Bool
    }

    state {
        T: Table(Bits(256), Bool)
    }

    oracle Sample() -> Bits(256) {
        nonce <-$ Bits(256) sample-name nonce;
        if bnonce {
            assert (T[nonce] == None);
            T[nonce] <- Some(true);
        }
        return nonce;
    }
}
package PRF {
    params {
        bprf: Bool,
        prf: fn Bits(256), Bits(256) -> Bits(256)
    }

    state {
        k: Maybe(Bits(256)),
        T: Table(Bits(256), Bits(256))
    }

    oracle Eval(m: Bits(256)) -> Bits(256) {
        if bprf {
            if (T[m] == None) {
                r <-$ Bits(256) sample-name r;
                T[m] <- Some(r);
            }
            return Unwrap(T[m]);
        }
        if (k == None) {
            r <-$ Bits(256) sample-name k;
            k <- Some(r);
        }
        return prf(Unwrap(k), m);
    }
}
package Reduction {
    params {
        bmod: Bool,
        xor: fn Bits(256), Bits(256) -> Bits(256)
    }

    import oracles {
        GetMessage(m: Bits(256)) -> Bits(256),
        Sample() -> Bits(256),
        Eval(x: Bits(256)) -> Bits(256),
        Xor(m: Bits(256)) -> Bits(256),
    }

    oracle Enc(m: Bits(256)) -> (Bits(256), Bits(256)) {
        nonce <- invoke Sample();
        m <- invoke GetMessage(m);
        if bmod {
            c <- invoke Xor(m);
            return (nonce, c);
        }
        pad <- invoke Eval(nonce);
        return (nonce, xor(pad, m));
    }
}

package XOR {
    params {
        bxor: Bool,
        xor: fn Bits(256), Bits(256) -> Bits(256)
    }

    oracle Xor(m: Bits(256)) -> Bits(256) {
        pad <-$ Bits(256) sample-name pad;
        if bxor {
            return pad;
        }
        return xor(pad, m);
    }
}
